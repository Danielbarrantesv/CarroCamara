   128→    /**
   129→     * Binds Preview + primary VideoCapture use case to the given lifecycleOwner and surfaces
   130→     * the preview onto [previewView]. If dual camera is enabled and the device supports it,
   131→     * a secondary VideoCapture is also bound for the other lens.
   132→     */
   133→    fun bindCamera(lifecycleOwner: LifecycleOwner, previewView: PreviewView) {
   134→        if (ContextCompat.checkSelfPermission(ctx, Manifest.permission.CAMERA)
   135→            != PackageManager.PERMISSION_GRANTED
   136→        ) {
   137→            _state.value = _state.value.copy(error = "Permiso de cámara no concedido")
   138→            return
   139→        }
   140→
   141→        // If recording is in progress, just re-attach the preview surface
   142→        // without disturbing the active recording. This is what lets the user
   143→        // switch tabs and come back without losing the recording.
   144→        if (_state.value.isRecording) {
   145→            try {
   146→                previewUseCase?.setSurfaceProvider(previewView.surfaceProvider)
   147→            } catch (_: Exception) {}
   148→            return
   149→        }
   150→
   151→        viewModelScope.launch {
   152→            try {
   153→                val provider = getCameraProvider(ctx)
   154→                cameraProvider = provider
   155→
   156→                val preview = Preview.Builder().build().also { p ->
   157→                    p.setSurfaceProvider(previewView.surfaceProvider)
   158→                }
   159→                previewUseCase = preview
   160→
   161→                val quality = settings.videoQuality.first()
   162→                val recorder = buildRecorder(quality)
   163→                val capture = VideoCapture.withOutput(recorder)
   164→
   165→                val selector = CameraSelector.Builder()
   166→                    .requireLensFacing(_state.value.lensFacing)
   167→                    .build()
   168→
   169→                provider.unbindAll()
   170→
   171→                // ===== Dual camera detection =====
   172→                val dualRequested = settings.dualCameraEnabled.first()
   173→                val concurrentSupported = try {
   174→                    provider.availableConcurrentCameraInfos.isNotEmpty()
   175→                } catch (_: Exception) { false }
   176→
   177→                if (dualRequested && !concurrentSupported) {
   178→                    _state.value = _state.value.copy(
   179→                        error = "Tu dispositivo no soporta grabación con doble cámara. " +
   180→                            "Se grabará solo con la cámara seleccionada."
   181→                    )
   182→                }
   183→
   184→                if (dualRequested && concurrentSupported) {
   185→                    // Use CameraX ConcurrentCamera API (CameraX 1.4+)
   186→                    val backSelector = if (_state.value.lensFacing == CameraSelector.LENS_FACING_BACK)
   187→                        CameraSelector.DEFAULT_BACK_CAMERA else CameraSelector.DEFAULT_FRONT_CAMERA
   188→                    val frontSelector = if (_state.value.lensFacing == CameraSelector.LENS_FACING_BACK)
   189→                        CameraSelector.DEFAULT_FRONT_CAMERA else CameraSelector.DEFAULT_BACK_CAMERA
   190→
   191→                    val backRecorder = buildRecorder(quality)
   192→                    val backCapture = VideoCapture.withOutput(backRecorder)
   193→                    val frontRecorder = buildRecorder(quality)
   194→                    val frontCapture = VideoCapture.withOutput(frontRecorder)
   195→
   196→                    val primaryCap: VideoCapture<Recorder>
   197→                    val secondaryCap: VideoCapture<Recorder>
   198→
   199→                    val backUseCaseGroup = UseCaseGroup.Builder()
   200→                        .addUseCase(preview)
   201→                        .addUseCase(backCapture)
   202→                        .build()
   203→                    val frontUseCaseGroup = UseCaseGroup.Builder()
   204→                        .addUseCase(frontCapture)
   205→                        .build()
   206→
   207→                    val backConfig = ConcurrentCamera.SingleCameraConfig(
   208→                        backSelector, backUseCaseGroup, lifecycleOwner
   209→                    )
   210→                    val frontConfig = ConcurrentCamera.SingleCameraConfig(
   211→                        frontSelector, frontUseCaseGroup, lifecycleOwner
   212→                    )
   213→
   214→                    try {
   215→                        provider.bindToLifecycle(listOf(backConfig, frontConfig))
   216→                        if (_state.value.lensFacing == CameraSelector.LENS_FACING_BACK) {
   217→                            primaryCap = backCapture
   218→                            secondaryCap = frontCapture
   219→                        } else {
   220→                            primaryCap = frontCapture
   221→                            secondaryCap = backCapture
   222→                        }
   223→                        primaryCapture = primaryCap
   224→                        secondaryCapture = secondaryCap
   225→                    } catch (e: Exception) {
   226→                        Log.w(TAG, "Concurrent camera bind failed, falling back to single", e)
   227→                        provider.bindToLifecycle(lifecycleOwner, backSelector, preview, backCapture)
   228→                        primaryCapture = backCapture
   229→                        secondaryCapture = null
   230→                    }
   231→                } else {
   232→                    provider.bindToLifecycle(lifecycleOwner, selector, preview, capture)
   233→                    primaryCapture = capture
   234→                    secondaryCapture = null
   235→                }
   236→
   237→                _state.value = _state.value.copy(isCameraBound = true)
   238→                startGpsStream()
   239→            } catch (e: Exception) {
   240→                Log.e(TAG, "bindCamera failed", e)
   241→                _state.value = _state.value.copy(
   242→                    error = "No se pudo abrir la cámara: ${e.message}",
   243→                    isCameraBound = false
   244→                )
   245→            }
   246→        }
   247→    }

   283→    private fun startRecording() {
   284→        val capture = primaryCapture ?: run {
   285→            _state.value = _state.value.copy(error = "Cámara no lista. Cierra y abre la app.")
   286→            return
   287→        }
   288→
   289→        viewModelScope.launch {
   290→            try {
   291→                val audioEnabled = settings.audioEnabled.first()
   292→                val dualEnabled = _state.value.dualCameraEnabled && secondaryCapture != null
   293→                val storageUriStr = settings.storageUri.first()
   294→                val storageTreeUri = storageUriStr?.let { Uri.parse(it) }
   295→
   296→                tripStartMs = System.currentTimeMillis()
   297→                currentTripId = tripRepo.startTrip(tripStartMs, _state.value.lat, _state.value.lon)
   298→                segmentIndex = 0
   299→
   300→                _state.value = _state.value.copy(
   301→                    isRecording = true,
   302→                    startedAt = tripStartMs,
   303→                    durationSec = 0,
   304→                    distanceMeters = 0.0,
   305→                    maxSpeedKmh = 0f,
   306→                    avgSpeedKmh = 0f,
   307→                    gForce = 0f,
   308→                    error = null,
   309→                    lastEventLabel = null
   310→                )
   311→
   312→                val primaryLens = lensLabel(_state.value.lensFacing)
   313→                startNextSegment(capture, primaryLens, audioEnabled, storageTreeUri)
   314→
   315→                if (dualEnabled) {
   316→                    val secondaryLens = lensLabel(
   317→                        if (_state.value.lensFacing == CameraSelector.LENS_FACING_BACK)
   318→                            CameraSelector.LENS_FACING_FRONT
   319→                        else CameraSelector.LENS_FACING_BACK
   320→                    )
   321→                    secondaryCapture?.let { secCap ->
   322→                        startSecondarySegment(secCap, secondaryLens, audioEnabled, storageTreeUri)
   323→                    }
   324→                }
   325→
   326→                startTimer()
   327→                startGForceStream()
   328→            } catch (e: Exception) {
   329→                Log.e(TAG, "startRecording failed", e)
   330→                _state.value = _state.value.copy(error = "No se pudo iniciar: ${e.message}")
   331→            }
   332→        }
   333→    }

   335→    private fun lensLabel(lens: Int): String =
   336→        if (lens == CameraSelector.LENS_FACING_BACK) "Back" else "Front"

   338→    private fun startNextSegment(
   339→        capture: VideoCapture<Recorder>,
   340→        lensLabel: String,
   341→        audioEnabled: Boolean,
   342→        storageTreeUri: Uri?
   343→    ) {
   344→        val tripId = currentTripId
   345→        val index = segmentIndex
   346→
   347→        viewModelScope.launch {
   348→            try {
   349→                val limitGb = settings.storageLimitGb.first()
   350→                val limitBytes = limitGb.toLong() * 1_073_741_824L
   351→                var used = tripRepo.totalVideoSizeBytes()
   352→                while (used > limitBytes) {
   353→                    val oldest = tripRepo.oldestUnprotectedSegment() ?: break
   354→                    FileUtils.safeDelete(File(oldest.filePath))
   355→                    tripRepo.deleteSegment(oldest.id)
   356→                    used = tripRepo.totalVideoSizeBytes()
   357→                }
   358→
   359→                val outFile = FileUtils.videoFileForSegment(ctx, tripId, index, lensLabel)
   360→                val outputOpts = FileOutputOptions.Builder(outFile).build()
   361→                val pending = capture.output.prepareRecording(ctx, outputOpts)
   362→
   363→                currentSegmentStartMs = System.currentTimeMillis()
   364→
   365→                primaryRecording = if (audioEnabled) {
   366→                    pending.withAudioEnabled()
   367→                        .start(ContextCompat.getMainExecutor(ctx)) { event ->
   368→                            handleVideoEvent(event, tripId, index, lensLabel, outFile, storageTreeUri, isSecondary = false)
   369→                        }
   370→                } else {
   371→                    pending.start(ContextCompat.getMainExecutor(ctx)) { event ->
   372→                        handleVideoEvent(event, tripId, index, lensLabel, outFile, storageTreeUri, isSecondary = false)
   373→                    }
   374→                }
   375→
   376→                _state.value = _state.value.copy(
   377→                    currentSegmentIndex = index,
   378→                    storageUsedBytes = used
   379→                )
   380→                segmentIndex++
   381→            } catch (e: Exception) {
   382→                Log.e(TAG, "startNextSegment failed", e)
   383→            }
   384→        }
   385→    }

   387→    private fun startSecondarySegment(
   388→        capture: VideoCapture<Recorder>,
   389→        lensLabel: String,
   390→        audioEnabled: Boolean,
   391→        storageTreeUri: Uri?
   392→    ) {
   393→        val tripId = currentTripId
   394→        val index = segmentIndex
   395→
   396→        viewModelScope.launch {
   397→            try {
   398→                val outFile = FileUtils.videoFileForSegment(ctx, tripId, index, lensLabel)
   399→                val outputOpts = FileOutputOptions.Builder(outFile).build()
   400→                val pending = capture.output.prepareRecording(ctx, outputOpts)
   401→
   402→                // Note: secondary camera cannot use audio (only one mic at a time)
   403→                secondaryRecording = pending.start(ContextCompat.getMainExecutor(ctx)) { event ->
   404→                    handleVideoEvent(event, tripId, index, lensLabel, outFile, storageTreeUri, isSecondary = true)
   405→                }
   406→            } catch (e: Exception) {
   407→                Log.e(TAG, "startSecondarySegment failed", e)
   408→            }
   409→        }
   410→    }

   412→    private fun handleVideoEvent(
   413→        event: VideoRecordEvent,
   414→        tripId: Long,
   415→        index: Int,
   416→        lensLabel: String,
   417→        outFile: File,
   418→        storageTreeUri: Uri?,
   419→        isSecondary: Boolean
   420→    ) {
   421→        when (event) {
   422→            is VideoRecordEvent.Finalize -> {
   423→                if (!event.hasError()) {
   424→                    viewModelScope.launch {
   425→                        val durationMs = System.currentTimeMillis() - currentSegmentStartMs
   426→                        // If a custom storage folder is set, move the file there
   427→                        val finalUri = FileUtils.moveToCustomStorage(
   428→                            ctx, outFile, storageTreeUri, outFile.name
   429→                        )
   430→                        val finalPath = finalUri?.toString() ?: outFile.absolutePath
   431→
   432→                        tripRepo.addSegment(
   433→                            VideoSegmentEntity(
   434→                                tripId = tripId,
   435→                                filePath = finalPath,
   436→                                createdAt = currentSegmentStartMs,
   437→                                durationMs = durationMs,
   438→                                sizeBytes = if (storageTreeUri != null) {
   439→                                    // File was moved to SAF, size is what we recorded
   440→                                    outFile.length().takeIf { it > 0 } ?: 0L
   441→                                } else outFile.length()
   442→                            )
   443→                        )
   444→                        if (_state.value.isRecording && !isSecondary) {
   445→                            val audio = settings.audioEnabled.first()
   446→                            // Restart only the primary segment
   447→                            primaryCapture?.let { cap ->
   448→                                startNextSegment(cap, lensLabel, audio, storageTreeUri)
   449→                            }
   450→                        }
   451→                    }
   452→                } else {
   453→                    Log.e(TAG, "Finalize error ($lensLabel): ${event.cause}")
   454→                    val cause = event.cause
   455→                    if (cause != null && _state.value.isRecording) {
   456→                        _state.value = _state.value.copy(
   457→                            error = "Grabación interrumpida: ${cause.message ?: "error desconocido"}"
   458→                        )
   459→                    }
   460→                }
   461→            }
   462→        }
   463→    }

   465→    private fun startTimer() {
   466→        timerJob?.cancel()
   467→        timerJob = viewModelScope.launch {
   468→            while (_state.value.isRecording) {
   469→                delay(1000)
   470→                _state.value = _state.value.copy(
   471→                    durationSec = (System.currentTimeMillis() - tripStartMs) / 1000
   472→                )
   473→            }
   474→        }
   475→    }

   477→    private fun startGpsStream() {
   478→        gpsJob?.cancel()
   479→        gpsJob = viewModelScope.launch {
   480→            var lastLat: Double? = null
   481→            var lastLon: Double? = null
   482→            var distance: Double = 0.0
   483→            var maxSpeed: Float = 0f
   484→            var speedSum: Float = 0f
   485→            var sampleCount: Int = 0
   486→
   487→            try {
   488→                gpsTracker.locationUpdates().collect { loc ->
   489→                    val speed = if (loc.hasSpeed()) loc.speed else 0f
   490→                    val speedKmh = speed * 3.6f
   491→
   492→                    if (lastLat != null && lastLon != null) {
   493→                        distance += haversine(lastLat!!, lastLon!!, loc.latitude, loc.longitude)
   494→                    }
   495→                    lastLat = loc.latitude
   496→                    lastLon = loc.longitude
   497→
   498→                    if (speedKmh > maxSpeed) maxSpeed = speedKmh
   499→                    speedSum += speedKmh
   500→                    sampleCount++
   501→
   502→                    if (_state.value.isRecording) {
   503→                        tripRepo.addGpsSample(
   504→                            GpsSampleEntity(
   505→                                tripId = currentTripId,
   506→                                createdAt = System.currentTimeMillis(),
   507→                                lat = loc.latitude,
   508→                                lon = loc.longitude,
   509→                                altitude = if (loc.hasAltitude()) loc.altitude else null,
   510→                                accuracy = if (loc.hasAccuracy()) loc.accuracy else null,
   511→                                speedMs = if (loc.hasSpeed()) loc.speed else null,
   512→                                bearing = if (loc.hasBearing()) loc.bearing else null
   513→                            )
   514→                        )
   515→                    }
   516→
   517→                    _state.value = _state.value.copy(
   518→                        speedKmh = speedKmh,
   519→                        distanceMeters = distance,
   520→                        lat = loc.latitude,
   521→                        lon = loc.longitude,
   522→                        maxSpeedKmh = maxSpeed,
   523→                        avgSpeedKmh = if (sampleCount > 0) speedSum / sampleCount else 0f
   524→                    )
   525→                }
   526→            } catch (e: Exception) {
   527→                Log.w(TAG, "GPS stream error: ${e.message}")
   528→            }
   529→        }
   530→    }

   532→    private fun startGForceStream() {
   533→        gForceJob?.cancel()
   534→        gForceJob = viewModelScope.launch {
   535→            val threshold = settings.gforceThreshold.first()
   536→            var peakG = 0f
   537→
   538→            try {
   539→                gForceDetector.samples().collect { sample ->
   540→                    if (sample.magnitudeG > peakG) {
   541→                        peakG = sample.magnitudeG
   542→                        _state.value = _state.value.copy(gForce = peakG)
   543→                    }
   544→
   545→                    val type = gForceDetector.classify(sample, threshold) ?: return@collect
   546→
   547→                    if (_state.value.isRecording) {
   548→                        tripRepo.addEvent(
   549→                            GForceEventEntity(
   550→                                tripId = currentTripId,
   551→                                createdAt = sample.timestampMs,
   552→                                type = type.name,
   553→                                magnitudeG = sample.magnitudeG,
   554→                                axisX = sample.x / GForceDetector.GRAVITY,
   555→                                axisY = sample.y / GForceDetector.GRAVITY,
   556→                                axisZ = sample.z / GForceDetector.GRAVITY,
   557→                                lat = _state.value.lat,
   558→                                lon = _state.value.lon,
   559→                                speedKmh = _state.value.speedKmh
   560→                            )
   561→                        )
   562→                        _state.value = _state.value.copy(
   563→                            lastEventLabel = "${type.label} · ${"%.2f".format(sample.magnitudeG)}g"
   564→                        )
   565→                    }
   566→                }
   567→            } catch (e: Exception) {
   568→                Log.w(TAG, "GForce stream error: ${e.message}")
   569→            }
   570→        }
   571→    }

   573→    private fun stopRecording() {
   574→        try { primaryRecording?.stop() } catch (_: Exception) {}
   575→        try { secondaryRecording?.stop() } catch (_: Exception) {}
   576→        primaryRecording = null
   577→        secondaryRecording = null
   578→        timerJob?.cancel()
   579→        gForceJob?.cancel()
   580→
   581→        viewModelScope.launch {
   582→            val endMs = System.currentTimeMillis()
   583→            val durationSec = (endMs - tripStartMs) / 1000
   584→            val st = _state.value
   585→
   586→            if (currentTripId > 0) {
   587→                tripRepo.endTrip(
   588→                    id = currentTripId,

   658→    private fun haversine(lat1: Double, lon1: Double, lat2: Double, lon2: Double): Double {
   659→        val r = 6_371_000.0
   660→        val dLat = Math.toRadians(lat2 - lat1)
   661→        val dLon = Math.toRadians(lon2 - lon1)
   662→        val a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
   663→            Math.cos(Math.toRadians(lat1)) * Math.cos(Math.toRadians(lat2)) *
   664→            Math.sin(dLon / 2) * Math.sin(dLon / 2)
   665→        val c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
   666→        return r * c
   667→    }
