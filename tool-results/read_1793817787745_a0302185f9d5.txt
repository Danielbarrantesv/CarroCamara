   128→   128→    /**
   129→   129→     * Binds Preview + primary VideoCapture use case to the given lifecycleOwner and surfaces
   130→   130→     * the preview onto [previewView]. If dual camera is enabled and the device supports it,
   131→   131→     * a secondary VideoCapture is also bound for the other lens.
   132→   132→     */
   133→   133→    fun bindCamera(lifecycleOwner: LifecycleOwner, previewView: PreviewView) {
   134→   134→        if (ContextCompat.checkSelfPermission(ctx, Manifest.permission.CAMERA)
   135→   135→            != PackageManager.PERMISSION_GRANTED
   136→   136→        ) {
   137→   137→            _state.value = _state.value.copy(error = "Permiso de cámara no concedido")
   138→   138→            return
   139→   139→        }
   140→   140→
   141→   141→        // If recording is in progress, just re-attach the preview surface
   142→   142→        // without disturbing the active recording. This is what lets the user
   143→   143→        // switch tabs and come back without losing the recording.
   144→   144→        if (_state.value.isRecording) {
   145→   145→            try {
   146→   146→                previewUseCase?.setSurfaceProvider(previewView.surfaceProvider)
   147→   147→            } catch (_: Exception) {}
   148→   148→            return
   149→   149→        }
   150→   150→
   151→   151→        viewModelScope.launch {
   152→   152→            try {
   153→   153→                val provider = getCameraProvider(ctx)
   154→   154→                cameraProvider = provider
   155→   155→
   156→   156→                val preview = Preview.Builder().build().also { p ->
   157→   157→                    p.setSurfaceProvider(previewView.surfaceProvider)
   158→   158→                }
   159→   159→                previewUseCase = preview
   160→   160→
   161→   161→                val quality = settings.videoQuality.first()
   162→   162→                val recorder = buildRecorder(quality)
   163→   163→                val capture = VideoCapture.withOutput(recorder)
   164→   164→
   165→   165→                val selector = CameraSelector.Builder()
   166→   166→                    .requireLensFacing(_state.value.lensFacing)
   167→   167→                    .build()
   168→   168→
   169→   169→                provider.unbindAll()
   170→   170→
   171→   171→                // ===== Dual camera detection =====
   172→   172→                val dualRequested = settings.dualCameraEnabled.first()
   173→   173→                val concurrentSupported = try {
   174→   174→                    provider.availableConcurrentCameraInfos.isNotEmpty()
   175→   175→                } catch (_: Exception) { false }
   176→   176→
   177→   177→                if (dualRequested && !concurrentSupported) {
   178→   178→                    _state.value = _state.value.copy(
   179→   179→                        error = "Tu dispositivo no soporta grabación con doble cámara. " +
   180→   180→                            "Se grabará solo con la cámara seleccionada."
   181→   181→                    )
   182→   182→                }
   183→   183→
   184→   184→                if (dualRequested && concurrentSupported) {
   185→   185→                    // Use CameraX ConcurrentCamera API (CameraX 1.4+)
   186→   186→                    val backSelector = if (_state.value.lensFacing == CameraSelector.LENS_FACING_BACK)
   187→   187→                        CameraSelector.DEFAULT_BACK_CAMERA else CameraSelector.DEFAULT_FRONT_CAMERA
   188→   188→                    val frontSelector = if (_state.value.lensFacing == CameraSelector.LENS_FACING_BACK)
   189→   189→                        CameraSelector.DEFAULT_FRONT_CAMERA else CameraSelector.DEFAULT_BACK_CAMERA
   190→   190→
   191→   191→                    val backRecorder = buildRecorder(quality)
   192→   192→                    val backCapture = VideoCapture.withOutput(backRecorder)
   193→   193→                    val frontRecorder = buildRecorder(quality)
   194→   194→                    val frontCapture = VideoCapture.withOutput(frontRecorder)
   195→   195→
   196→   196→                    val primaryCap: VideoCapture<Recorder>
   197→   197→                    val secondaryCap: VideoCapture<Recorder>
   198→   198→
   199→   199→                    val backUseCaseGroup = UseCaseGroup.Builder()
   200→   200→                        .addUseCase(preview)
   201→   201→                        .addUseCase(backCapture)
   202→   202→                        .build()
   203→   203→                    val frontUseCaseGroup = UseCaseGroup.Builder()
   204→   204→                        .addUseCase(frontCapture)
   205→   205→                        .build()
   206→   206→
   207→   207→                    val backConfig = ConcurrentCamera.SingleCameraConfig(
   208→   208→                        backSelector, backUseCaseGroup, lifecycleOwner
   209→   209→                    )
   210→   210→                    val frontConfig = ConcurrentCamera.SingleCameraConfig(
   211→   211→                        frontSelector, frontUseCaseGroup, lifecycleOwner
   212→   212→                    )
   213→   213→
   214→   214→                    try {
   215→   215→                        provider.bindToLifecycle(listOf(backConfig, frontConfig))
   216→   216→                        if (_state.value.lensFacing == CameraSelector.LENS_FACING_BACK) {
   217→   217→                            primaryCap = backCapture
   218→   218→                            secondaryCap = frontCapture
   219→   219→                        } else {
   220→   220→                            primaryCap = frontCapture
   221→   221→                            secondaryCap = backCapture
   222→   222→                        }
   223→   223→                        primaryCapture = primaryCap
   224→   224→                        secondaryCapture = secondaryCap
   225→   225→                    } catch (e: Exception) {
   226→   226→                        Log.w(TAG, "Concurrent camera bind failed, falling back to single", e)
   227→   227→                        provider.bindToLifecycle(lifecycleOwner, backSelector, preview, backCapture)
   228→   228→                        primaryCapture = backCapture
   229→   229→                        secondaryCapture = null
   230→   230→                    }
   231→   231→                } else {
   232→   232→                    provider.bindToLifecycle(lifecycleOwner, selector, preview, capture)
   233→   233→                    primaryCapture = capture
   234→   234→                    secondaryCapture = null
   235→   235→                }
   236→   236→
   237→   237→                _state.value = _state.value.copy(isCameraBound = true)
   238→   238→                startGpsStream()
   239→   239→            } catch (e: Exception) {
   240→   240→                Log.e(TAG, "bindCamera failed", e)
   241→   241→                _state.value = _state.value.copy(
   242→   242→                    error = "No se pudo abrir la cámara: ${e.message}",
   243→   243→                    isCameraBound = false
   244→   244→                )
   245→   245→            }
   246→   246→        }
   247→   247→    }

   283→   283→    private fun startRecording() {
   284→   284→        val capture = primaryCapture ?: run {
   285→   285→            _state.value = _state.value.copy(error = "Cámara no lista. Cierra y abre la app.")
   286→   286→            return
   287→   287→        }
   288→   288→
   289→   289→        viewModelScope.launch {
   290→   290→            try {
   291→   291→                val audioEnabled = settings.audioEnabled.first()
   292→   292→                val dualEnabled = _state.value.dualCameraEnabled && secondaryCapture != null
   293→   293→                val storageUriStr = settings.storageUri.first()
   294→   294→                val storageTreeUri = storageUriStr?.let { Uri.parse(it) }
   295→   295→
   296→   296→                tripStartMs = System.currentTimeMillis()
   297→   297→                currentTripId = tripRepo.startTrip(tripStartMs, _state.value.lat, _state.value.lon)
   298→   298→                segmentIndex = 0
   299→   299→
   300→   300→                _state.value = _state.value.copy(
   301→   301→                    isRecording = true,
   302→   302→                    startedAt = tripStartMs,
   303→   303→                    durationSec = 0,
   304→   304→                    distanceMeters = 0.0,
   305→   305→                    maxSpeedKmh = 0f,
   306→   306→                    avgSpeedKmh = 0f,
   307→   307→                    gForce = 0f,
   308→   308→                    error = null,
   309→   309→                    lastEventLabel = null
   310→   310→                )
   311→   311→
   312→   312→                val primaryLens = lensLabel(_state.value.lensFacing)
   313→   313→                startNextSegment(capture, primaryLens, audioEnabled, storageTreeUri)
   314→   314→
   315→   315→                if (dualEnabled) {
   316→   316→                    val secondaryLens = lensLabel(
   317→   317→                        if (_state.value.lensFacing == CameraSelector.LENS_FACING_BACK)
   318→   318→                            CameraSelector.LENS_FACING_FRONT
   319→   319→                        else CameraSelector.LENS_FACING_BACK
   320→   320→                    )
   321→   321→                    secondaryCapture?.let { secCap ->
   322→   322→                        startSecondarySegment(secCap, secondaryLens, audioEnabled, storageTreeUri)
   323→   323→                    }
   324→   324→                }
   325→   325→
   326→   326→                startTimer()
   327→   327→                startGForceStream()
   328→   328→            } catch (e: Exception) {
   329→   329→                Log.e(TAG, "startRecording failed", e)
   330→   330→                _state.value = _state.value.copy(error = "No se pudo iniciar: ${e.message}")
   331→   331→            }
   332→   332→        }
   333→   333→    }

   338→   338→    private fun startNextSegment(
   339→   339→        capture: VideoCapture<Recorder>,
   340→   340→        lensLabel: String,
   341→   341→        audioEnabled: Boolean,
   342→   342→        storageTreeUri: Uri?
   343→   343→    ) {
   344→   344→        val tripId = currentTripId
   345→   345→        val index = segmentIndex
   346→   346→
   347→   347→        viewModelScope.launch {
   348→   348→            try {
   349→   349→                val limitGb = settings.storageLimitGb.first()
   350→   350→                val limitBytes = limitGb.toLong() * 1_073_741_824L
   351→   351→                var used = tripRepo.totalVideoSizeBytes()
   352→   352→                while (used > limitBytes) {
   353→   353→                    val oldest = tripRepo.oldestUnprotectedSegment() ?: break
   354→   354→                    FileUtils.safeDelete(File(oldest.filePath))
   355→   355→                    tripRepo.deleteSegment(oldest.id)
   356→   356→                    used = tripRepo.totalVideoSizeBytes()
   357→   357→                }
   358→   358→
   359→   359→                val outFile = FileUtils.videoFileForSegment(ctx, tripId, index, lensLabel)
   360→   360→                val outputOpts = FileOutputOptions.Builder(outFile).build()
   361→   361→                val pending = capture.output.prepareRecording(ctx, outputOpts)
   362→   362→
   363→   363→                currentSegmentStartMs = System.currentTimeMillis()
   364→   364→
   365→   365→                primaryRecording = if (audioEnabled) {
   366→   366→                    pending.withAudioEnabled()
   367→   367→                        .start(ContextCompat.getMainExecutor(ctx)) { event ->
   368→   368→                            handleVideoEvent(event, tripId, index, lensLabel, outFile, storageTreeUri, isSecondary = false)
   369→   369→                        }
   370→   370→                } else {
   371→   371→                    pending.start(ContextCompat.getMainExecutor(ctx)) { event ->
   372→   372→                        handleVideoEvent(event, tripId, index, lensLabel, outFile, storageTreeUri, isSecondary = false)
   373→   373→                    }
   374→   374→                }
   375→   375→
   376→   376→                _state.value = _state.value.copy(
   377→   377→                    currentSegmentIndex = index,
   378→   378→                    storageUsedBytes = used
   379→   379→                )
   380→   380→                segmentIndex++
   381→   381→            } catch (e: Exception) {
   382→   382→                Log.e(TAG, "startNextSegment failed", e)
   383→   383→            }
   384→   384→        }
   385→   385→    }

   387→   387→    private fun startSecondarySegment(
   388→   388→        capture: VideoCapture<Recorder>,
   389→   389→        lensLabel: String,
   390→   390→        audioEnabled: Boolean,
   391→   391→        storageTreeUri: Uri?
   392→   392→    ) {
   393→   393→        val tripId = currentTripId
   394→   394→        val index = segmentIndex
   395→   395→
   396→   396→        viewModelScope.launch {
   397→   397→            try {
   398→   398→                val outFile = FileUtils.videoFileForSegment(ctx, tripId, index, lensLabel)
   399→   399→                val outputOpts = FileOutputOptions.Builder(outFile).build()
   400→   400→                val pending = capture.output.prepareRecording(ctx, outputOpts)
   401→   401→
   402→   402→                // Note: secondary camera cannot use audio (only one mic at a time)
   403→   403→                secondaryRecording = pending.start(ContextCompat.getMainExecutor(ctx)) { event ->
   404→   404→                    handleVideoEvent(event, tripId, index, lensLabel, outFile, storageTreeUri, isSecondary = true)
   405→   405→                }
   406→   406→            } catch (e: Exception) {
   407→   407→                Log.e(TAG, "startSecondarySegment failed", e)
   408→   408→            }
   409→   409→        }
   410→   410→    }
   411→   411→
   412→   412→    private fun handleVideoEvent(
   413→   413→        event: VideoRecordEvent,
   414→   414→        tripId: Long,
   415→   415→        index: Int,
   416→   416→        lensLabel: String,
   417→   417→        outFile: File,
   418→   418→        storageTreeUri: Uri?,
   419→   419→        isSecondary: Boolean
   420→   420→    ) {
   421→   421→        when (event) {
   422→   422→            is VideoRecordEvent.Finalize -> {
   423→   423→                if (!event.hasError()) {
   424→   424→                    viewModelScope.launch {
   425→   425→                        val durationMs = System.currentTimeMillis() - currentSegmentStartMs
   426→   426→                        // If a custom storage folder is set, move the file there
   427→   427→                        val finalUri = FileUtils.moveToCustomStorage(
   428→   428→                            ctx, outFile, storageTreeUri, outFile.name
   429→   429→                        )
   430→   430→                        val finalPath = finalUri?.toString() ?: outFile.absolutePath
   431→   431→
   432→   432→                        tripRepo.addSegment(
   433→   433→                            VideoSegmentEntity(
   434→   434→                                tripId = tripId,
   435→   435→                                filePath = finalPath,
   436→   436→                                createdAt = currentSegmentStartMs,
   437→   437→                                durationMs = durationMs,
   438→   438→                                sizeBytes = if (storageTreeUri != null) {
   439→   439→                                    // File was moved to SAF, size is what we recorded
   440→   440→                                    outFile.length().takeIf { it > 0 } ?: 0L
   441→   441→                                } else outFile.length()
   442→   442→                            )
   443→   443→                        )
   444→   444→                        if (_state.value.isRecording && !isSecondary) {
   445→   445→                            val audio = settings.audioEnabled.first()
   446→   446→                            // Restart only the primary segment
   447→   447→                            primaryCapture?.let { cap ->
   448→   448→                                startNextSegment(cap, lensLabel, audio, storageTreeUri)
   449→   449→                            }
   450→   450→                        }
   451→   451→                    }
   452→   452→                } else {
   453→   453→                    Log.e(TAG, "Finalize error ($lensLabel): ${event.cause}")
   454→   454→                    val cause = event.cause
   455→   455→                    if (cause != null && _state.value.isRecording) {
   456→   456→                        _state.value = _state.value.copy(
   457→   457→                            error = "Grabación interrumpida: ${cause.message ?: "error desconocido"}"
   458→   458→                        )
   459→   459→                    }
   460→   460→                }
   461→   461→            }
   462→   462→        }
   463→   463→    }
   464→   464→
   465→   465→    private fun startTimer() {
   466→   466→        timerJob?.cancel()
   467→   467→        timerJob = viewModelScope.launch {
   468→   468→            while (_state.value.isRecording) {
   469→   469→                delay(1000)
   470→   470→                _state.value = _state.value.copy(
   471→   471→                    durationSec = (System.currentTimeMillis() - tripStartMs) / 1000
   472→   472→                )
   473→   473→            }
   474→   474→        }
   475→   475→    }

   477→   477→    private fun startGpsStream() {
   478→   478→        gpsJob?.cancel()
   479→   479→        gpsJob = viewModelScope.launch {
   480→   480→            var lastLat: Double? = null
   481→   481→            var lastLon: Double? = null
   482→   482→            var distance: Double = 0.0
   483→   483→            var maxSpeed: Float = 0f
   484→   484→            var speedSum: Float = 0f
   485→   485→            var sampleCount: Int = 0
   486→   486→
   487→   487→            try {
   488→   488→                gpsTracker.locationUpdates().collect { loc ->
   489→   489→                    val speed = if (loc.hasSpeed()) loc.speed else 0f
   490→   490→                    val speedKmh = speed * 3.6f
   491→   491→
   492→   492→                    if (lastLat != null && lastLon != null) {
   493→   493→                        distance += haversine(lastLat!!, lastLon!!, loc.latitude, loc.longitude)
   494→   494→                    }
   495→   495→                    lastLat = loc.latitude
   496→   496→                    lastLon = loc.longitude
   497→   497→
   498→   498→                    if (speedKmh > maxSpeed) maxSpeed = speedKmh
   499→   499→                    speedSum += speedKmh
   500→   500→                    sampleCount++
   501→   501→
   502→   502→                    if (_state.value.isRecording) {
   503→   503→                        tripRepo.addGpsSample(
   504→   504→                            GpsSampleEntity(
   505→   505→                                tripId = currentTripId,
   506→   506→                                createdAt = System.currentTimeMillis(),
   507→   507→                                lat = loc.latitude,
   508→   508→                                lon = loc.longitude,
   509→   509→                                altitude = if (loc.hasAltitude()) loc.altitude else null,
   510→   510→                                accuracy = if (loc.hasAccuracy()) loc.accuracy else null,
   511→   511→                                speedMs = if (loc.hasSpeed()) loc.speed else null,
   512→   512→                                bearing = if (loc.hasBearing()) loc.bearing else null
   513→   513→                            )
   514→   514→                        )
   515→   515→                    }
   516→   516→
   517→   517→                    _state.value = _state.value.copy(
   518→   518→                        speedKmh = speedKmh,
   519→   519→                        distanceMeters = distance,
   520→   520→                        lat = loc.latitude,
   521→   521→                        lon = loc.longitude,
   522→   522→                        maxSpeedKmh = maxSpeed,
   523→   523→                        avgSpeedKmh = if (sampleCount > 0) speedSum / sampleCount else 0f
   524→   524→                    )
   525→   525→                }
   526→   526→            } catch (e: Exception) {
   527→   527→                Log.w(TAG, "GPS stream error: ${e.message}")
   528→   528→            }
   529→   529→        }
   530→   530→    }

   658→   658→    private fun haversine(lat1: Double, lon1: Double, lat2: Double, lon2: Double): Double {
   659→   659→        val r = 6_371_000.0
   660→   660→        val dLat = Math.toRadians(lat2 - lat1)
   661→   661→        val dLon = Math.toRadians(lon2 - lon1)
   662→   662→        val a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
   663→   663→            Math.cos(Math.toRadians(lat1)) * Math.cos(Math.toRadians(lat2)) *
   664→   664→            Math.sin(dLon / 2) * Math.sin(dLon / 2)
   665→   665→        val c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
   666→   666→        return r * c
   667→   667→    }
